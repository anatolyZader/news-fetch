## Your task

Regenerate `docs/INDEX.md` — a one-directional compiled index of the entire `docs/` tree with a one-line description, last-commit date, and health flags per file. **Write only `docs/INDEX.md` and (if missing) one pointer line in `docs/README.md`. **Step 1 — Inventory (deterministic)**

Run this node script from `/home/eventstorm1/news`:

```
node -e "
const fs = require('fs');
const path = require('path');
const { execSync } = require('child_process');
const root = '/home/eventstorm1/news';
const docsRoot = path.join(root, 'docs');

function walk(dir) {
  return fs.readdirSync(dir, { withFileTypes: true }).flatMap(e => {
    const p = path.join(dir, e.name);
    if (e.isDirectory()) return e.name === 'node_modules' ? [] : walk(p);
    return [p];
  });
}

const files = walk(docsRoot).sort();
const out = files.map(p => {
  const rel = path.relative(root, p);
  const st = fs.statSync(p);
  let lastCommit = null;
  try {
    lastCommit = execSync('git log -1 --format=%as -- ' + JSON.stringify(rel), { cwd: root }).toString().trim() || null;
  } catch (e) {}
  const isMd = p.endsWith('.md');
  let firstHeading = null, mdLinks = [];
  if (isMd) {
    const text = fs.readFileSync(p, 'utf8');
    const h = text.match(/^#{1,2} +(.+)$/m);
    firstHeading = h ? h[1].trim() : null;
    mdLinks = [...text.matchAll(/\]\(([^)#\s]+\.md)(#[^)]*)?\)/g)].map(m => {
      const target = m[1];
      if (/^https?:/.test(target)) return null;
      const resolved = path.resolve(path.dirname(p), target);
      return { target, resolves: fs.existsSync(resolved) };
    }).filter(Boolean);
  }
  return {
    path: rel,
    dir: path.relative(docsRoot, path.dirname(p)) || '(top-level)',
    sizeKB: Math.round(st.size / 102.4) / 10,
    lastCommit,
    firstHeading,
    brokenLinks: mdLinks.filter(l => !l.resolves).map(l => l.target),
    linkCount: mdLinks.length,
    isMd,
  };
});
console.log(JSON.stringify({ total: out.length, files: out }, null, 2));
"
```
